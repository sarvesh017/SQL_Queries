7. display the name, deptname of top five earning employees.

1..using orderby + limit +offset
Select e.ename ,d.deptname,e.salary  from emp e join dept d on e.deptid = d.deptid order by e.salary desc limit 5;

2.using ROW_NUMBER()-

Select ename ,salary,deptname from (select e.ename,d.deptname,e.salary ,row_number() over (order by e.salary desc ) as rnk from emp e join dept d on e.deptid = d.deptid) as result where rnk<=5;

3.using rank

SELECT ename, deptname, salary
FROM (
    SELECT e.ename,
           d.deptname,
           e.salary,
           RANK() OVER (ORDER BY e.salary DESC) AS rnk
    FROM emp e
    JOIN dept d ON e.deptid = d.deptid
) AS result
WHERE rnk <= 5;

4 .correlated subquery-

SELECT e.ename, d.deptname, e.salary
FROM emp e
JOIN dept d ON e.deptid = d.deptid
WHERE (
    SELECT COUNT(DISTINCT e2.salary)
    FROM emp e2
    WHERE e2.salary > e.salary
) < 5
ORDER BY e.salary DESC;
